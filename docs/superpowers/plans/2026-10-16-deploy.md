## 決定（2026-10-16，角色 1）

1. **`start.bat` 很薄**：只找 `.venv\Scripts\python.exe`（沒有就說明並暫停），然後執行 `python -m src.api.launch`；出錯時暫停讓人看得到訊息。`launch` 依序：沒有 `.env` 就從 `.env.example` 建一個並填入新的 `JWT_SECRET`（已經有 `.env` 就完全不動）→ 執行 seed（同 `python -m src.api.seed`）→ 背景等 `/api/health` 回應後打開 `http://127.0.0.1:8000/web/` → 在前景用 uvicorn 跑 `src.api.main:app`（`127.0.0.1:8000`，不加 `--reload`）。
2. **新設定**：`CATEGORY_MODEL_DIR`（預設 `models/category`，相對路徑從 repo 根目錄算）、`CATEGORY_MODEL_REPO`、`CATEGORY_MODEL_REVISION`（預設空白）。設了 `REPO` 就一定要設 `REVISION`（固定版本），否則啟動時停下來並指出變數名稱。
3. **`CATEGORY_MODEL=bert`**：模型資料夾沒有 `config.json`，或 `src/ml/category.py` 還不存在時，分類模型是 `unavailable`（健康檢查顯示 `unavailable`、即時分類 503、上傳任務 failed），服務照樣啟動；兩者都在時才接上真的模型（那一步是任務 2.15）。`ATTRIBUTES_MODEL` 這批不動。
4. **容器啟動**（`python -m src.api.container`）：沒有 `JWT_SECRET` 時用資料 volume 裡的 `/data/jwt-secret`（第一次啟動時產生），重建容器後舊的 token 照樣有效；`CATEGORY_MODEL=bert`、模型資料夾沒有 `config.json`、也設了 `REPO`／`REVISION` 時，用 `huggingface_hub.snapshot_download` 下載到暫存資料夾，完整下載後才改名成模型資料夾（下載到一半失敗不會留下半套模型）；下載失敗只印一行原因，服務照樣啟動；`DEMO_MODE=1` 而且資料庫裡一個使用者都沒有時匯入假用戶；最後用 uvicorn 聽 `0.0.0.0:8000`。
5. **映像檔**：`python:3.12-slim`，只 COPY `requirements*.txt`、`configs/`、`src/`；`.dockerignore` 另外擋掉 `rawdata/`、`data/`、`*.csv`、壓縮檔、`.env*`、`models/`、`runtime/`、`experiments/`、`src/web/`（v1 的建置來源，不提供，而且可能有本機產生的 `*-data.js`）。`huggingface_hub` 放在 `requirements-docker.txt`，只有映像檔裝。CPU 版 `torch`、`transformers` 等角色 2 交出 BERT（任務 3.6）時再加進 `requirements-docker.txt`。
6. **compose**：一個服務，兩個 volume（`db:/data`、`models:/models`），`DEMO_MODE=1`，`LLAMA_SERVER_URL` 預設 `http://host.docker.internal:8080`（主機上沒有 llama-server 就是無 GPU 模式），埠只開在 `127.0.0.1:8000`。`CATEGORY_MODEL` 預設 `fake`，角色 2 公開模型（任務 3.9）後改成 `bert` 並填上 `REPO`／`REVISION`。

```python
"""What start.bat runs: a .env with a fresh secret when there is none, the demo users, the v2 pages opened in the
browser once the server answers, and the server itself on this computer only. Run from the repository root:

    .venv/Scripts/python.exe -m src.api.launch
"""
import secrets
import threading
import time
import urllib.request
import webbrowser

from .db import ROOT

HOST, PORT = "127.0.0.1", 8000  # this computer only: DEMO_MODE lets anyone log in as a demo user
PAGE = f"http://{HOST}:{PORT}/web/"
HEALTH = f"http://{HOST}:{PORT}/api/health"


def ensure_env(root):
    """Write root/.env from .env.example with a new JWT_SECRET when there is no .env; True when it did. An existing
    .env is never read or changed."""
    env = root / ".env"
    if env.exists():
        return False
    lines = (root / ".env.example").read_text(encoding="utf-8").splitlines()
    secret = secrets.token_urlsafe(32)
    env.write_text("".join((f"JWT_SECRET={secret}" if line.startswith("JWT_SECRET=") else line) + "\n" for line in lines),
                   encoding="utf-8")
    return True


def answers(url):
    """Whether url answers 200, straight to it: a system proxy would otherwise take 127.0.0.1 too."""
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    try:
        with opener.open(url, timeout=1) as response:
            return response.status == 200
    except OSError:
        return False


def open_when_ready(url=PAGE, health=HEALTH, opener=webbrowser.open, fetch=answers, timeout=60, pause=0.5):
    """Open url once health answers; give up quietly after timeout seconds. True when the page was opened."""
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        if fetch(health):
            opener(url)
            return True
        time.sleep(pause)
    return False


def open_in_background():
    threading.Thread(target=open_when_ready, name="open-browser", daemon=True).start()


def seed_demo():
    from . import seed  # reads .env itself, so only after ensure_env
    seed.main()


def main(root=ROOT):
    if ensure_env(root):
        print("已建立 .env（含一組新的 JWT_SECRET）；要改設定請編輯 .env 後重新啟動")
    seed_demo()
    open_in_background()
    import uvicorn
    uvicorn.run("src.api.main:app", host=HOST, port=PORT)


if __name__ == "__main__":
    main()
```

```python
"""What the Docker container runs (design D14): the secret, the category model when one is to be downloaded, the
demo users on an empty database, then the server on all of the container's interfaces (docker-compose.yml publishes
the port on the host's 127.0.0.1 only).

    python -m src.api.container
"""
import logging
import os
import secrets
import shutil
from pathlib import Path

from sqlmodel import Session, select

from .backends import model_dir
from .db import init_db, make_engine
from .settings import Settings
from .tables import User

DATA = Path("/data")  # the database volume
log = logging.getLogger(__name__)


def jwt_secret(env, data_dir=DATA):
    """JWT_SECRET from the environment, or the one kept in the data volume, made on the first start, so tokens keep
    working when the container is rebuilt."""
    given = env.get("JWT_SECRET", "").strip()
    if given:
        return given
    kept = data_dir / "jwt-secret"
    if not kept.exists():
        data_dir.mkdir(parents=True, exist_ok=True)
        kept.write_text(secrets.token_urlsafe(32), encoding="utf-8")
    return kept.read_text(encoding="utf-8").strip()


def hub_download(repo_id, revision, local_dir):
    from huggingface_hub import snapshot_download  # only the image installs it
    snapshot_download(repo_id=repo_id, revision=revision, local_dir=local_dir)


def ensure_category_model(settings, download=hub_download):
    """Download the category model when CATEGORY_MODEL=bert and its folder holds none. It goes to a temporary folder
    renamed only when complete, so a failure leaves no half model and the next start tries again. A failure is one
    line in the log: the server starts anyway and reports the model unavailable."""
    if settings.category_model != "bert":
        return "skipped"
    folder = model_dir(settings)
    if (folder / "config.json").exists():
        return "present"
    if not settings.category_model_repo:
        log.warning("CATEGORY_MODEL=bert, no model in %s and no CATEGORY_MODEL_REPO to download it from", folder)
        return "no_repo"
    partial = folder.with_name(folder.name + ".partial")
    shutil.rmtree(partial, ignore_errors=True)
    try:
        download(settings.category_model_repo, settings.category_model_revision, str(partial))
    except Exception as error:  # no network, a wrong repo or revision: start without the model
        shutil.rmtree(partial, ignore_errors=True)
        log.warning("could not download the category model %s@%s: %s", settings.category_model_repo,
                    settings.category_model_revision, error)
        return "failed"
    shutil.rmtree(folder, ignore_errors=True)
    partial.rename(folder)
    return "downloaded"


def seed_if_empty(engine, demo_mode):
    """The demo users, only in demo mode and only into a database nobody is in yet; how many were added."""
    if not demo_mode:
        return 0
    with Session(engine) as session:
        if session.exec(select(User.id).limit(1)).first() is not None:
            return 0
    from ..receipt.personas import build_personas
    from ..receipt.traits import load_context
    from .db import ROOT
    from .seed import seed_demo_users
    return seed_demo_users(engine, build_personas(load_context(ROOT)["calendar"]))


def main():
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(name)s: %(message)s")
    os.environ["JWT_SECRET"] = jwt_secret(os.environ)
    settings = Settings.from_env(os.environ)
    ensure_category_model(settings)
    engine = make_engine(settings.database_url)
    init_db(engine)
    added = seed_if_empty(engine, settings.demo_mode)
    if added:
        log.info("added %d demo users", added)
    engine.dispose()
    import uvicorn
    uvicorn.run("src.api.main:app", host="0.0.0.0", port=8000)


if __name__ == "__main__":
    main()
```
