## 決定（2026-10-16，角色 1）

1. **`months`**：這個人有品項或有上傳任務的月份，由舊到新（`accounts.months_of`）。只有任務、沒有留下品項的月份（例如整個月的明細都被略過）也算，刪月份才找得到它。
2. **顯示名稱**的規則和註冊時一樣：去掉前後空白後 1–20 字元，不能有控制字元、看不見的字元或換行（`validation.DisplayName`，註冊也改用它）。`username` 不能改：`PATCH` 帶了也不理。
3. **新密碼**和註冊一樣是 8–128 字元（`security.Password`）。目前密碼錯回 403 `wrong_password`，什麼都不改。
4. **登出、改密碼**都用 `security.end_sessions(user)` 換掉 `token_key`。假用戶是共用的：用假用戶登出，所有切換成同一個假用戶的人都會被登出（5.11 寫明登出照常可用）。
5. **刪帳號**（D13）：一個交易裡依序刪 `items` → `jobs` → `profiles` → `users`，再刪 `item_cache` 裡「這個人的品項用過、現在已經沒有任何品項用到」的品名（每個屬性模型的答案都刪）。只在即時分類框查過、從沒上傳過的品名沒有記錄是誰查的，不屬於任何人，不會因為刪帳號被刪。還有 `queued` 或 `running` 的任務回 409 `job_in_progress`。
6. **刪月份**：`month` 要寫成 `YYYY-MM`（月份 01–12），否則 422；那個月沒有品項也沒有任務回 404 `month_not_found`；只有**那個月**有 `queued` 或 `running` 的任務才回 409 `job_in_progress`（別的月份的任務不擋）。刪掉那個月的 `items` 和 `jobs`，清快取的方式同第 5 點，然後把重算排進 app 的背景執行緒（`app.state.worker.call`），排在已經排隊的上傳任務後面，不在回應請求的執行緒上算。重算跑完之前，`/api/me/profile` 的 `ready`、`item_count`、`areas`、`traits` 還是舊的（`months` 立刻是新的）。重算還在排隊時帳號被刪掉，重算就什麼都不做。
7. **`jobs.Worker.call(label, task)`**：上傳任務照舊用 `add(job_id)`（紀錄裡寫 `upload job N`），其他工作用 `call`，出錯時用 `label` 寫進紀錄，下一個照常跑。
8. `DELETE /api/me` 的密碼放在 JSON body（`{"password": "…"}`），和 5.11 一樣。

```python
def run_recompute(engine, backends, context, user_id):
    """Work a user's profile out again on the worker, as after a month of theirs was deleted. An account deleted while
    this waited has nothing left to recompute."""
    with Session(engine) as session:
        if session.get(User, user_id) is None:
            return
        profiles.recompute(session, backends, context, user_id)
```

`Worker` 的 `add`、`loop` 改成，並加上 `call`：

```python
    def add(self, job_id):
        """Queue an upload's job."""
        self.call(f"upload job {job_id}", partial(self.run, job_id))

    def call(self, label, task):
        """Queue other work that writes profiles, such as the recompute after a month is deleted. It waits behind the
        uploads queued before it, so two writes of one profile never overlap; label names it in the log."""
        self.queue.put((label, task))

    def loop(self):
        while True:
            label, task = self.queue.get()
            try:
                task()
            # run_job already fails a job that crashes; this keeps anything else from ending the loop, which would
            # leave every later job queued until a restart.
            except Exception:
                log.exception("%s failed", label)
            finally:
                self.queue.task_done()
```

（`Worker` 的 docstring 補一句：上傳以外，刪月份之後的重算也排在這裡。）

```python
"""What a user can delete: a month of their uploads or the whole account, and with it the attribute answers no stored
item needs any more. Nothing here commits; the route does, so each deletion is one transaction."""
from sqlmodel import col, delete, select

from .jobs import UNFINISHED
from .tables import Item, ItemCache, Job, Profile, User


def months_of(session, user_id):
    """The months the user uploaded, oldest first: those with items, and those with a job even if no line was kept."""
    with_items = session.exec(select(Item.month).where(Item.user_id == user_id).distinct()).all()
    with_jobs = session.exec(select(Job.month).where(Job.user_id == user_id).distinct()).all()
    return sorted(set(with_items) | set(with_jobs))


def busy(session, user_id, month=None):
    """Whether the user has an upload job queued or running, of that month when one is given."""
    query = select(Job.id).where(Job.user_id == user_id, col(Job.status).in_(UNFINISHED))
    if month is not None:
        query = query.where(Job.month == month)
    return session.exec(query.limit(1)).first() is not None


def forget_unused(session, names):
    """Delete every model's answers for those of names that no stored item has any more. A name only the classify box
    was asked about is nobody's, so it is never among the names a deletion passes here."""
    if not names:
        return
    still_used = select(Item.name).where(col(Item.name).in_(names))
    session.exec(delete(ItemCache).where(col(ItemCache.name).in_(names), col(ItemCache.name).not_in(still_used)))


def delete_month(session, user_id, month):
    """Delete one month's items and jobs, then the answers only they used."""
    names = list(set(session.exec(select(Item.name).where(Item.user_id == user_id, Item.month == month)).all()))
    session.exec(delete(Item).where(Item.user_id == user_id, Item.month == month))
    session.exec(delete(Job).where(Job.user_id == user_id, Job.month == month))
    forget_unused(session, names)


def delete_account(session, user_id):
    """Delete the user and everything they uploaded, in the order the foreign keys need (items, jobs, profile, then
    the user), then the answers only their items used."""
    names = list(set(session.exec(select(Item.name).where(Item.user_id == user_id)).all()))
    session.exec(delete(Item).where(Item.user_id == user_id))
    session.exec(delete(Job).where(Job.user_id == user_id))
    session.exec(delete(Profile).where(Profile.user_id == user_id))
    session.exec(delete(User).where(User.id == user_id))
    forget_unused(session, names)
```

```python
"""Account management (overview 5.11): see and rename the account, change its password, delete it. Logout is in
routes/auth.py; deleting a month is below too."""
from fastapi import APIRouter, Request, Response
from pydantic import BaseModel

from .. import accounts
from ..db import SessionDep
from ..errors import ApiError
from ..security import Password, UserDep, end_sessions, hash_password, token_response, verify_password
from ..validation import DisplayName

router = APIRouter(prefix="/api/me")


class Rename(BaseModel):
    display_name: DisplayName


class PasswordChange(BaseModel):
    current_password: str
    new_password: Password


class Confirmation(BaseModel):
    password: str


def writable(user):
    """Demo users are shared by everyone, so nobody may change or delete one; checked before any password."""
    if user.is_demo:
        raise ApiError(403, "demo_user_read_only")


def confirm(user, password):
    if not verify_password(password, user.password_hash):
        raise ApiError(403, "wrong_password")


def account(session, user):
    return {"user_id": user.id, "username": user.username, "display_name": user.display_name,
            "is_demo": user.is_demo, "months": accounts.months_of(session, user.id)}


@router.get("")
def my_account(user: UserDep, session: SessionDep):
    return account(session, user)


@router.patch("")
def rename(body: Rename, user: UserDep, session: SessionDep):
    writable(user)
    user.display_name = body.display_name
    session.add(user)
    session.commit()
    return account(session, user)


@router.post("/password")
def change_password(body: PasswordChange, user: UserDep, request: Request, session: SessionDep):
    writable(user)
    confirm(user, body.current_password)
    user.password_hash = hash_password(body.new_password)
    end_sessions(user)  # the answer's token carries the new key, so this device stays logged in
    session.add(user)
    session.commit()
    return token_response(user, request.app.state.settings)


@router.delete("", status_code=204)
def delete_account(body: Confirmation, user: UserDep, session: SessionDep):
    writable(user)
    confirm(user, body.password)
    # A job still to run would write the user's names into the cache again after they are gone.
    if accounts.busy(session, user.id):
        raise ApiError(409, "job_in_progress")
    accounts.delete_account(session, user.id)
    session.commit()
    return Response(status_code=204)
```

`src/api/app.py`：`from .routes import account, auth, ...`，`app.include_router(me.router)` 後面加 `app.include_router(account.router)`。

```python
Month = Annotated[str, Path(pattern=r"^\d{4}-(0[1-9]|1[0-2])$")]  # YYYY-MM


@router.delete("/months/{month}", status_code=204)
def delete_month(month: Month, user: UserDep, request: Request, session: SessionDep):
    writable(user)
    if month not in accounts.months_of(session, user.id):
        raise ApiError(404, "month_not_found")
    if accounts.busy(session, user.id, month):
        raise ApiError(409, "job_in_progress")
    accounts.delete_month(session, user.id, month)
    session.commit()
    state = request.app.state
    # Not here on the request's thread: an upload job of this user may be writing the profile right now. Queued behind
    # it on the worker, the recompute reads the items as they are when it runs.
    state.worker.call(f"recompute for user {user.id}",
                      partial(jobs.run_recompute, state.engine, state.backends, state.context, user.id))
    return Response(status_code=204)
```
