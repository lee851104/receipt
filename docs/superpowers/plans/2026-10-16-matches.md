```python
"""The match list: everyone comparable with me scored now from the stored vectors, the people past the bar on one
side, and how far from me they shop. Nothing here reads the database; the route hands it the profiles."""
from ..receipt import places, similarity

# The farthest places.distance each scope allows; None is any distance, an unknown one included.
SCOPES = {"all": None, "district": 0, "city": 1, "region": 2}


def entries(mine, my_areas, people, model, regions):
    """Everyone who can be compared with me, in the order given: their score, its whole percentage, their home
    district closest to mine, and how far apart we shop (None when either of us has no home district)."""
    scored = []
    for person in people:
        result = similarity.compare(mine, person["vector"], model)
        if not result["comparable"]:
            continue
        areas = person["areas"]
        scored.append({"user_id": person["user_id"], "display_name": person["display_name"],
                       "score": result["score"], "percent": similarity.percent(result["score"]),
                       "area": places.closest_area(my_areas, areas, regions),
                       "distance": places.distance(my_areas, areas, regions) if my_areas and areas else None})
    return scored


def pick(entries, kind, level, match):
    """The people past the bar on one side, how many of them shop within level of me, and the first few of those.

    The bar is checked on the whole percentage the page prints, as src/web/match-filter.js does, so a card reading
    +30% is always listed. Similar goes from the highest score and opposite from the lowest; ties go by display
    name in code-point order, then by user id."""
    opposite = kind == "opposite"
    line = similarity.percent(-match["opposite_score"] if opposite else match["min_score"])
    eligible = [entry for entry in entries if (entry["percent"] <= line if opposite else entry["percent"] >= line)]
    within = [entry for entry in eligible
              if level is None or (entry["distance"] is not None and entry["distance"] <= level)]
    within.sort(key=lambda entry: (entry["score"] if opposite else -entry["score"],
                                   entry["display_name"], entry["user_id"]))
    return {"eligible": len(eligible), "within": len(within), "people": within[:match["top"]]}
```

def stored_vector(profile, size):
    """A profile's taste vector, or None without a profile or with too few items for one. A vector of another length
    was stored before configs/traits.json changed: fail (a 500) rather than read its values into the wrong traits."""
    if profile is None or profile.vector is None:
        return None
    vector = json.loads(profile.vector)
    if len(vector) != size:
        raise ValueError("a stored taste vector has the wrong length; work every profile out again")
    return vector

```python
"""GET /api/matches: the people whose taste is closest to mine, or most opposite, within a distance of me."""
import json
from typing import Literal

from fastapi import APIRouter, Request
from sqlmodel import col, select

from ...receipt.traits import similarity_model
from .. import matching, profiles
from ..db import SessionDep
from ..errors import ApiError
from ..security import UserDep
from ..tables import Profile, User

router = APIRouter(prefix="/api/matches")


# A plain def, not async: the reads block, so FastAPI runs this in a worker thread.
@router.get("")
def match_list(user: UserDep, request: Request, session: SessionDep,
               kind: Literal["similar", "opposite"] = "similar",
               scope: Literal["all", "district", "city", "region"] = "all"):
    state = request.app.state
    settings = state.context["traits"]
    size = len(settings["traits"])
    profile = session.get(Profile, user.id)
    vector = profiles.stored_vector(profile, size)
    my_areas = [] if profile is None else json.loads(profile.areas)
    head = {"kind": kind, "scope": scope, "ready": vector is not None, "my_areas": my_areas}
    # Too little data of my own comes first: whatever the scope, there is no list to narrow, so no error either.
    if vector is None:
        return {**head, "eligible": 0, "within": 0, "people": []}
    level = matching.SCOPES[scope]
    if level is not None and not my_areas:
        raise ApiError(400, "no_home_area")
    # Scores are worked out now from everyone's stored profile and never stored (design D3).
    others = session.exec(select(User, Profile).join(Profile, col(Profile.user_id) == col(User.id))
                          .where(col(User.id) != user.id, col(Profile.vector).is_not(None))).all()
    people = [{"user_id": other.id, "display_name": other.display_name,
               "vector": profiles.stored_vector(stored, size), "areas": json.loads(stored.areas)}
              for other, stored in others]
    entries = matching.entries(vector, my_areas, people, similarity_model(settings), state.regions)
    return {**head, **matching.pick(entries, kind, level, settings["match"])}
```
