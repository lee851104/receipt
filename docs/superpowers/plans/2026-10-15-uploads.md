```python
"""Parse invoice exports into classified records with coded invoice IDs."""

import calendar
import csv
from datetime import datetime

from .address import parse_district

REQUIRED = ("發票日期", "發票狀態", "發票號碼", "賣方名稱", "消費明細_品名", "消費明細_數量", "消費明細_金額")
MIN_AMOUNT = 10  # an uploaded line below this (a discount, a free item, small change) is not analysed


def whole(text):
    """The integer a CSV cell holds, or None for a decimal such as 4.49; text that is no number raises ValueError."""
    try:
        return int(text)
    except ValueError:
        float(text)
        return None


def build_month(path, catalog, for_upload=False):
    """One month of an invoice export as report rows.

    An upload (for_upload=True) follows the invoice-ingestion rules: every required column must be there, voided
    invoices (a status starting with 作廢) are skipped, lines under MIN_AMOUNT or with a decimal amount are skipped,
    and a decimal quantity (litres of fuel, weighed goods) counts as 1. Without it the v1 report reads exactly as
    before."""
    rows, invoice_ids = [], {}
    month_key = None
    with path.open(encoding="utf-8-sig", newline="") as source:
        reader = csv.DictReader(source)
        if for_upload:
            missing = [column for column in REQUIRED if column not in (reader.fieldnames or [])]
            if missing:
                raise ValueError("Missing column: " + ", ".join(missing))
        for row in reader:
            if not row.get("發票日期"):
                continue  # CSV export footer, not a transaction.
            if for_upload and row["發票狀態"].startswith("作廢"):
                continue  # A voided invoice was never a purchase.
            if row["發票狀態"] != "開立已確認":
                raise ValueError("Unreviewed invoice status: " + row["發票狀態"])
            date = datetime.strptime(row["發票日期"], "%Y%m%d")
            key = date.strftime("%Y-%m")
            if month_key and month_key != key:
                raise ValueError("Expected one month per CSV")
            month_key = key
            if for_upload:
                amount, quantity = whole(row["消費明細_金額"]), whole(row["消費明細_數量"])
                if amount is None or amount < MIN_AMOUNT:
                    continue
                if quantity is None:
                    quantity = 1
            else:
                amount, quantity = int(row["消費明細_金額"]), int(row["消費明細_數量"])
            name = row["消費明細_品名"].strip()
            category = catalog.get(name, {"category": 10, "provisional": True})
            original_id = row["發票號碼"]
            invoice_ids.setdefault(original_id, f"{key}-R{len(invoice_ids) + 1:02}")
            rows.append({
                "day": date.day,
                "invoice": invoice_ids[original_id],
                "merchant": row["賣方名稱"],
                "district": parse_district(row.get("賣方地址")),
                "name": name,
                "quantity": quantity,
                "amount": amount,
                **category,
            })
    if not rows:
        raise ValueError("No invoice records")
    year, month = map(int, month_key.split("-"))
    return month_key, {
        "year": year, "monthNumber": month, "month": f"{year} 年 {month} 月",
        "days": calendar.monthrange(year, month)[1], "source": path.name,
        "rows": sorted(rows, key=lambda row: row["day"]),
    }
```

```python
"""Database URLs: tests share one in-memory connection, and a file always lives under the repository root."""
import tempfile
import threading
import unittest
from pathlib import Path

from sqlmodel import Session, select

from src.api.db import ROOT, file_url, init_db, make_engine
from src.api.tables import User


def add_amy(engine):
    with Session(engine) as session:
        session.add(User(username="amy", display_name="Amy"))
        session.commit()


def usernames(engine):
    with Session(engine) as session:
        return session.exec(select(User.username)).all()


class DatabaseTests(unittest.TestCase):
    def test_a_relative_file_is_kept_under_the_repository_root(self):
        self.assertEqual(file_url("sqlite:///data/app.db"), "sqlite:///" + (ROOT / "data" / "app.db").as_posix())

    def test_an_absolute_file_stays_where_it_is_and_its_folder_is_made(self):
        with tempfile.TemporaryDirectory() as folder:
            path = Path(folder) / "nested" / "app.db"
            engine = make_engine("sqlite:///" + path.as_posix())
            try:
                init_db(engine)
                add_amy(engine)
                self.assertEqual(usernames(engine), ["amy"])
            finally:
                engine.dispose()  # Windows cannot delete a file that is still open, even after a failed check
            self.assertTrue(path.is_file())

    def test_a_file_database_writes_ahead_and_waits_when_busy(self):
        with tempfile.TemporaryDirectory() as folder:
            engine = make_engine("sqlite:///" + (Path(folder) / "app.db").as_posix())
            try:
                with engine.connect() as connection:
                    mode = connection.exec_driver_sql("PRAGMA journal_mode").scalar()
                    timeout = connection.exec_driver_sql("PRAGMA busy_timeout").scalar()
            finally:
                engine.dispose()  # Windows cannot delete a file that is still open, even after a failed check
        self.assertEqual((mode, timeout), ("wal", 5000))

    def test_both_in_memory_spellings_share_one_database_across_threads(self):
        for url in ("sqlite://", "sqlite:///:memory:"):
            with self.subTest(url=url):
                engine = make_engine(url)
                init_db(engine)
                # FastAPI runs plain def routes in worker threads, so another thread must see the same tables.
                thread = threading.Thread(target=add_amy, args=(engine,))
                thread.start()
                thread.join()
                self.assertEqual(usernames(engine), ["amy"])


if __name__ == "__main__":
    unittest.main()
```

```python
"""Database tables: accounts, upload jobs, the purchases they hold, the attribute cache and taste profiles."""
import secrets
from datetime import datetime, timezone

from sqlmodel import Field, SQLModel


def now():
    return datetime.now(timezone.utc)


class User(SQLModel, table=True):
    __tablename__ = "users"
    id: int | None = Field(default=None, primary_key=True)
    username: str = Field(unique=True, index=True)
    password_hash: str | None = None  # demo users have none, so only demo-login lets them in
    display_name: str
    is_demo: bool = False
    persona: str | None = None        # the fictional type, such as 咖啡上班族; None for real people
    # Goes into every token next to the id. SQLite reuses the id of a deleted user, and a fresh database starts
    # the ids again, but a new account gets a new key, so an old token never works for it.
    token_key: str = Field(default_factory=lambda: secrets.token_hex(8))


class Job(SQLModel, table=True):
    """One upload: queued, then running, then done or failed; done and total count distinct product names."""
    __tablename__ = "jobs"
    # Never reuse the id of a deleted job: a page still polling the old id must not see a different upload.
    __table_args__ = {"sqlite_autoincrement": True}
    id: int | None = Field(default=None, primary_key=True)
    user_id: int = Field(foreign_key="users.id", index=True)
    month: str                        # YYYY-MM
    status: str = "queued"
    done: int = 0
    total: int = 0
    message: str = "排隊中"
    created_at: datetime = Field(default_factory=now)


class Item(SQLModel, table=True):
    """One purchase line kept from an upload. Only what the analysis needs: no address, no invoice number."""
    __tablename__ = "items"
    id: int | None = Field(default=None, primary_key=True)
    user_id: int = Field(foreign_key="users.id", index=True)
    month: str = Field(index=True)    # YYYY-MM
    day: int
    invoice: str                      # the coded YYYY-MM-Rnn, never the real invoice number
    merchant: str
    district: str | None = None       # city or county plus district only
    name: str
    quantity: int
    amount: int
    category: int = 10                # 0-9, or 10 (待確認) until classified or when the model is unsure
    provisional: bool = True          # True while category is 10
```

```python
"""The SQLite engine, and one session per request."""
from pathlib import Path
from typing import Annotated

from fastapi import Depends, Request
from sqlalchemy import event
from sqlalchemy.pool import StaticPool
from sqlmodel import Session, SQLModel, create_engine

from . import tables  # noqa: F401  registers every table on SQLModel.metadata

ROOT = Path(__file__).resolve().parents[2]
MEMORY = "sqlite://"


def file_url(url):
    """A SQLite file URL whose relative path starts at the repository root rather than wherever the server was
    started, so data/app.db always lands in the git-ignored data/ folder. Absolute paths stay as they are."""
    return "sqlite:///" + (ROOT / url.removeprefix("sqlite:///")).as_posix()


def write_ahead(connection, record):
    """Readers no longer wait for a background upload's writes, and a busy database is waited on for 5 seconds
    instead of failing at once."""
    cursor = connection.cursor()
    cursor.execute("PRAGMA journal_mode=WAL")
    cursor.execute("PRAGMA busy_timeout=5000")
    cursor.close()


def make_engine(url):
    """An engine for url. An in-memory database (for tests only) keeps one shared connection, or each request
    would see an empty one; that single connection cannot serve requests running at the same time."""
    if url in (MEMORY, "sqlite:///:memory:"):
        return create_engine(url, connect_args={"check_same_thread": False}, poolclass=StaticPool)
    if url.startswith("sqlite:///"):
        url = file_url(url)
        Path(url.removeprefix("sqlite:///")).parent.mkdir(parents=True, exist_ok=True)
    engine = create_engine(url, connect_args={"check_same_thread": False})
    event.listen(engine, "connect", write_ahead)
    return engine


def init_db(engine):
    """Create the tables that do not exist yet. It never changes a table that exists, so after a column is added
    to tables.py, delete data/app.db and run the seed again."""
    SQLModel.metadata.create_all(engine)


def get_session(request: Request):
    with Session(request.app.state.engine) as session:
        yield session


SessionDep = Annotated[Session, Depends(get_session)]
```

```python
"""Attributes through the item_cache table, so each product name is judged once per attribute model."""
import json

from sqlmodel import Session, col, select

from .tables import ItemCache


def model_ready(backends):
    """Whether the attribute model can judge a name now: the fake always, Qwen only while its llama-server answers."""
    return backends.attributes == "fake" or backends.gpu_probe()


def describe(session, backends, names, ask_model):
    """Each name's 23 attribute fields, or None while it is not analysed yet. The cache answers first; for the rest
    the model is asked (only its own cache unless ask_model and the model is ready), and new answers are stored."""
    rows = session.exec(select(ItemCache).where(ItemCache.model == backends.attributes,
                                                col(ItemCache.name).in_(names))).all()
    known = {row.name: json.loads(row.fields) for row in rows}
    missing = [name for name in dict.fromkeys(names) if name not in known]
    if missing:
        answers = backends.describe_items(missing, ask_model=ask_model and model_ready(backends))
        for name, fields in zip(missing, answers):
            if fields is not None:
                known[name] = fields
                # merge, not add: another upload may have stored the same name a moment ago
                session.merge(ItemCache(name=name, model=backends.attributes, fields=json.dumps(fields)))
        session.commit()
    return [known.get(name) for name in names]


def describe_later(engine, backends, name):
    """Judge one name in the background, for the classify box: its answer is waiting in the cache next time."""
    with Session(engine) as session:
        describe(session, backends, [name], ask_model=True)
```

```python
    session.add(job)
    session.commit()


def classify_items(session, job, backends, min_prob, names):
    """Store each name's category on the job's items: the model's first choice, or 10 below min_prob."""
    chosen = {}
    for name, answer in zip(names, backends.classify(names)):
        category = answer["category"]
        chosen[name] = category if answer["probs"][category] >= min_prob else PENDING
    items = session.exec(select(Item).where(Item.user_id == job.user_id, Item.month == job.month,
                                            col(Item.name).in_(list(chosen)))).all()
    # Only these names: a newer upload of the same month may have replaced the items meanwhile; its own job
    # classifies those.
    for item in items:
        item.category = chosen[item.name]
        item.provisional = item.category == PENDING
        session.add(item)
    session.commit()


def process(session, job, backends, settings):
    names = sorted(set(session.exec(select(Item.name).where(Item.user_id == job.user_id,
                                                            Item.month == job.month)).all()))
    update(session, job, status="running", done=0, total=len(names), message="分類品名")
    if backends.category == "unavailable":
        update(session, job, status="failed", message="分類模型無法使用，請稍後重新上傳")
        return
    classify_items(session, job, backends, settings.category_min_prob, names)
    update(session, job, message="判斷品項屬性")
    for start in range(0, len(names), BATCH):
        batch = names[start:start + BATCH]
        # A name the model cannot judge now (no GPU) stays pending, yet counts as handled.
        cache.describe(session, backends, batch, ask_model=True)
        update(session, job, done=job.done + len(batch))
    update(session, job, status="done", message="完成")


def run_job(engine, backends, settings, job_id):
    """Background task for one upload. Any crash marks the job failed instead of leaving it running."""
    with Session(engine) as session:
        job = session.get(Job, job_id)
        try:
            process(session, job, backends, settings)
        except Exception:
            log.exception("upload job %s failed", job_id)
            session.rollback()
            update(session, session.get(Job, job_id), status="failed", message="處理失敗，請重新上傳")


def fail_unfinished(engine):
    """At start-up nothing is running any more: jobs left queued or running by a stop are failed, to be uploaded
    again."""
    with Session(engine) as session:
        for job in session.exec(select(Job).where(col(Job.status).in_(UNFINISHED))).all():
            job.status, job.message = "failed", "服務重新啟動，請重新上傳"
            session.add(job)
        session.commit()
```

```python
"""Uploads: POST /api/invoices parses a month's CSV at once, then a background job classifies it;
GET /api/jobs/{job_id} tells how far that job is."""
import csv
import os
import tempfile
from pathlib import Path
from typing import Annotated

from fastapi import APIRouter, BackgroundTasks, Depends, Request, UploadFile
from sqlmodel import col, delete

from ...receipt.invoices import build_month
from ..db import SessionDep
from ..errors import ApiError
from ..jobs import UNFINISHED, run_job
from ..security import current_user
from ..tables import Item, Job, User
from ..validation import PathId

router = APIRouter(prefix="/api")
UserDep = Annotated[User, Depends(current_user)]
MAX_BYTES = 2 * 1024 * 1024


def parse(data):
    """The month key and rows of an uploaded CSV. The bytes go to a temporary file only while they are read."""
    handle = tempfile.NamedTemporaryFile(suffix=".csv", delete=False)
    try:
        with handle:
            handle.write(data)
        return build_month(Path(handle.name), {}, for_upload=True)
    finally:
        os.unlink(handle.name)


@router.post("/invoices", status_code=202)
def upload(file: UploadFile, user: UserDep, request: Request, background: BackgroundTasks, session: SessionDep):
    # Demo users are shared by everyone: an upload would hang one person's purchases on a public persona.
    if user.is_demo:
        raise ApiError(403, "demo_user_read_only")
    data = file.file.read(MAX_BYTES + 1)
    if len(data) > MAX_BYTES:
        raise ApiError(413, "file_too_large")
    try:
        month_key, month = parse(data)
    # ValueError covers UnicodeDecodeError, bad dates and numbers and every rule build_month enforces; csv.Error
    # a NUL byte or a broken quote.
    except (ValueError, csv.Error) as error:
        raise ApiError(400, "invalid_csv", str(error)) from None
    # The new file replaces the month: its old items go, and so do its finished jobs.
    session.exec(delete(Item).where(Item.user_id == user.id, Item.month == month_key))
    session.exec(delete(Job).where(Job.user_id == user.id, Job.month == month_key, col(Job.status).not_in(UNFINISHED)))
    for row in month["rows"]:
        session.add(Item(user_id=user.id, month=month_key, day=row["day"], invoice=row["invoice"],
                         merchant=row["merchant"], district=row["district"], name=row["name"],
                         quantity=row["quantity"], amount=row["amount"]))
    job = Job(user_id=user.id, month=month_key, total=len({row["name"] for row in month["rows"]}))
    session.add(job)
    session.commit()
    session.refresh(job)
    state = request.app.state
    background.add_task(run_job, state.engine, state.backends, state.settings, job.id)
    return {"job_id": job.id, "month": month_key, "rows": len(month["rows"])}


@router.get("/jobs/{job_id}")
def job_status(job_id: PathId, user: UserDep, session: SessionDep):
    job = session.get(Job, job_id)
    # Someone else's job answers exactly like a missing one, so ids tell nothing about other users.
    if job is None or job.user_id != user.id:
        raise ApiError(404, "not_found")
    return {"job_id": job.id, "month": job.month, "status": job.status, "done": job.done, "total": job.total,
            "message": job.message}
```

```python
"""POST /api/items/classify: one product name's category and attributes, for the classify box."""
from typing import Annotated

from fastapi import APIRouter, BackgroundTasks, Depends, Request
from pydantic import BaseModel, StringConstraints, field_validator

from .. import cache
from ..db import SessionDep
from ..errors import ApiError
from ..security import current_user
from ..validation import visible

router = APIRouter(prefix="/api/items", dependencies=[Depends(current_user)])


class ItemName(BaseModel):
    name: Annotated[str, StringConstraints(strip_whitespace=True, min_length=1, max_length=100)]

    @field_validator("name")
    @classmethod
    def visible_name(cls, name):
        return visible(name, "name")


# A plain def, not async: the models block, so FastAPI runs this in a worker thread.
@router.post("/classify")
def classify(body: ItemName, request: Request, background: BackgroundTasks, session: SessionDep):
    state = request.app.state
    backends = state.backends
    if backends.category == "unavailable":
        raise ApiError(503, "model_unavailable")
    [answer] = backends.classify([body.name])
    # The box never waits for Qwen: only the caches answer, and a name not seen yet is pending...
    [attributes] = cache.describe(session, backends, [body.name], ask_model=False)
    # ...while the model, if it can answer now, judges it in the background for next time.
    if attributes is None and cache.model_ready(backends):
        background.add_task(cache.describe_later, state.engine, backends, body.name)
    category = answer["category"]
    return {"name": body.name,
            "category": {"id": category, "name": state.categories[category], "prob": answer["probs"][category]},
            "probs": answer["probs"], "attributes": attributes,
            "attributes_status": "pending" if attributes is None else "ready"}
```
